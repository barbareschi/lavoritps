<script>
// Array globali per salvare i dati
var listaProdotti = [];
var carrello = [];

// Funzione che parte appena carica la pagina
window.onload = function() {
    var stringaCsv = localStorage.getItem('dati_csv');
    
    if (stringaCsv == null) {
        document.getElementById('spazioProdotti').innerHTML = "<h3>Attenzione: non hai caricato nessun file CSV da ini.html!</h3>";
        return;
    }

    faiParsingDelCsv(stringaCsv);
    disegnaProdottiSuSchermo();
};

// Funzione scolastica per fare il parsing del testo CSV
function faiParsingDelCsv(testoCompleto) {
    var righe = testoCompleto.split('\n');
    // La riga 0 contiene le intestazioni (marca,modello,ecc...)
    var intestazioni = righe[0].trim().split(',');

    // Parto da i=1 per saltare la prima riga di intestazione
    for (var i = 1; i < righe.length; i++) {
        var rigaCorrente = righe[i].trim();
        if (rigaCorrente == "") continue; // Salta le righe vuote

        var colonne = rigaCorrente.split(',');

        // Creo l'oggetto del prodotto
        var prod = {
            id: i - 1,
            marca: colonne[0],
            modello: colonne[1],
            descrizione: colonne[2],
            immagine: colonne[3],
            prezzo: parseFloat(colonne[4])
        };
        listaProdotti.push(prod);
    }
}

// Funzione che genera la grafica replicabile per ogni elemento
function disegnaProdottiSuSchermo() {
    var divContenitore = document.getElementById('spazioProdotti');
    divContenitore.innerHTML = ""; // Pulisco tutto

    for (var i = 0; i < listaProdotti.length; i++) {
        var p = listaProdotti[i];
        
        // Creo il box del prodotto singolo
        var box = document.createElement('div');
        box.className = "scheda-prodotto";
        box.innerHTML = `
            <img src="${p.immagine}">
            <h3>${p.marca}</h3>
            <p>${p.modello}</p>
            <div class="prezzo">€${p.prezzo.toFixed(2)}</div>
            <button class="btn-compra" onclick="mettiNelCarrello(${p.id})">Aggiungi</button>
            <button onclick="apriDettagliOggetto(${p.id})">Vedi Scheda</button>
        `;
        divContenitore.appendChild(box);
    }
}

// Funzione per la pagina dedicata alle caratteristiche dell'oggetto
function apriDettagliOggetto(idCercato) {
    // Cerco il prodotto nell'array usando un ciclo for classico
    var prod = null;
    for(var i=0; i<listaProdotti.length; i++) {
        if(listaProdotti[i].id == idCercato) {
            var prod = listaProdotti[i];
        }
    }

    var contenitore = document.getElementById('contenutoDettaglio');
    contenitore.innerHTML = `
        <div class="info-box">
            <div>
                <img src="${prod.immagine}" style="width:100%; max-width:400px; border: 1px solid #ccc;">
            </div>
            <div>
                <h1>${prod.marca} - ${prod.modello}</h1>
                <p style="font-size: 1.3em; margin: 20px 0; color: #555;"><strong>Specifiche:</strong> ${prod.descrizione}</p>
                <h2 style="color: green;">Prezzo di listino: €${prod.prezzo.toFixed(2)}</h2>
                <br><br>
                <button class="btn-compra" style="width: auto; padding: 15px 30px; font-size: 1.1em;" onclick="mettiNelCarrello(${prod.id}); chiudiSchermataDettaglio();">Metti nel carrello ed esci</button>
            </div>
        </div>
    `;

    document.getElementById('paginaDettaglio').style.display = 'block';
}

function chiudiSchermataDettaglio() {
    document.getElementById('paginaDettaglio').style.display = 'none';
}

// Funzione per gestire il riempimento del Basket
function mettiNelCarrello(idCercato) {
    var p = null;
    for(var i=0; i<listaProdotti.length; i++) {
        if(listaProdotti[i].id == idCercato) { p = listaProdotti[i]; }
    }

    // Controllo se c'è già nel carrello per aumentare la quantità
    var trovato = false;
    for (var j = 0; j < carrello.length; j++) {
        if (carrello[j].prodotto.id == idCercato) {
            carrello[j].quantita++;
            trovato = true;
        }
    }

    // Se non c'è lo aggiungo come elemento nuovo
    if (trovato == false) {
        carrello.push({ prodotto: p, quantita: 1 });
    }

    aggiornaGraficaCarrello();
}

function rimuoviDalCarrello(idCercato) {
    // Filtro l'array togliendo l'elemento
    var nuovoCarrello = [];
    for(var i=0; i<carrello.length; i++) {
        if(carrello[i].prodotto.id != idCercato) {
            nuovoCarrello.push(carrello[i]);
        }
    }
    carrello = nuovoCarrello;
    aggiornaGraficaCarrello();
}

function aggiornaGraficaCarrello() {
    var divCarrello = document.getElementById('elementiCarrello');
    var spanTotale = document.getElementById('prezzoTotale');
    var btnStampa = document.getElementById('bottoneStampa');

    if (carrello.length == 0) {
        divCarrello.innerHTML = "<p>Nessun oggetto nel carrello.</p>";
        spanTotale.innerText = "0.00";
        btnStampa.disabled = true;
        return;
    }

    divCarrello.innerHTML = "";
    var totaleSomma = 0;

    for (var i = 0; i < carrello.length; i++) {
        var item = carrello[i];
        var costoParziale = item.prodotto.prezzo * item.quantita;
        totaleSomma = totaleSomma + costoParziale;

        var rigaItem = document.createElement('div');
        rigaItem.style.marginBottom = "10px";
        rigaItem.innerHTML = `
            <strong>${item.prodotto.marca}</strong> - ${item.prodotto.modello} <br>
            ${item.quantita}x €${item.prodotto.prezzo.toFixed(2)} = <b>€${costoParziale.toFixed(2)}</b>
            <button onclick="rimuoviDalCarrello(${item.prodotto.id})" style="color:red; float:right;">X</button>
            <div style="clear:both;"></div>
        `;
        divCarrello.appendChild(rigaItem);
    }

    spanTotale.innerText = totaleSomma.toFixed(2);
    btnStampa.disabled = false; // Attivo il bottone per fare il PDF
}

// Funzione finale per stampare il PDF online usando jsPDF
function generaRicevutaPDF() {
    // Inizializzo jsPDF come da documentazione della libreria
    var { jsPDF } = window.jspdf;
    var doc = new jsPDF();

    // Scrittura dei testi sul foglio PDF (coordinate X, Y)
    doc.setFontSize(20);
    doc.text("RICEVUTA DEL PROCESSO DI ACQUISTO", 20, 20);
    
    doc.setFontSize(12);
    doc.text("E-commerce Universale - Progetto scolastico di Informatica", 20, 30);
    doc.text("Data di acquisto: " + new Date().toLocaleDateString(), 20, 37);
    doc.text("------------------------------------------------------------------------------------------", 20, 43);

    var altezzaY = 55;
    var sommaFinale = 0;

    doc.text("Articolo", 20, altezzaY);
    doc.text("Q.ta", 120, altezzaY);
    doc.text("Prezzo Unitario", 140, altezzaY);
    doc.text("Subtotale", 175, altezzaY);
    doc.text("------------------------------------------------------------------------------------------", 20, altezzaY+5);
    
    altezzaY = altezzaY + 15;

    // Ciclo for per scrivere i prodotti acquistati dentro il PDF
    for (var i = 0; i < carrello.length; i++) {
        var item = carrello[i];
        var sub = item.prodotto.prezzo * item.quantita;
        sommaFinale = sommaFinale + sub;

        var testoProdotto = item.prodotto.marca + " " + item.prodotto.modello;

        doc.text(testoProdotto, 20, altezzaY);
        doc.text(item.quantita.toString(), 122, altezzaY);
        doc.text("€" + item.prodotto.prezzo.toFixed(2), 140, altezzaY);
        doc.text("€" + sub.toFixed(2), 175, altezzaY);

        altezzaY = altezzaY + 10;
    }

    doc.text("------------------------------------------------------------------------------------------", 20, altezzaY);
    altezzaY = altezzaY + 10;

    doc.setFontSize(14);
    doc.text("TOTALE DA PAGARE: €" + sommaFinale.toFixed(2), 115, altezzaY);

    // Faccio scaricare il PDF al browser
    doc.save("ricevuta_carrello_pc.pdf");

    // Messaggio di conferma e azzeramento del basket
    alert("Grazie per aver acquistato! Il file PDF è stato generato e scaricato.");
    carrello = [];
    aggiornaGraficaCarrello();
}
</script>
