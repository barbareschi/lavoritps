<script>
function salvaTutto() {
    var inputInput = document.getElementById('fileCsv');
    var inputCss = document.getElementById('percorsoCss').value;

    // Controllo se il file è stato inserito
    if (inputInput.files.length == 0) {
        alert("Errore: Devi selezionare un file CSV!");
        return;
    }

    var file = inputInput.files[0];
    var lettore = new FileReader();

    // Leggo il file di testo
    lettore.onload = function(evento) {
        var contenutoCsv = evento.target.result;
        
        // Salvo nel localStorage così l'altra pagina può leggerlo
        localStorage.setItem('dati_csv', contenutoCsv);
        
        if(inputCss != "") {
            localStorage.setItem('foglio_css', inputCss);
        } else {
            localStorage.setItem('foglio_css', 'default.css');
        }

        alert("Impostazioni salvate! Adesso ti sposto al sito.");
        window.location.href = 'index.html'; // Cambio pagina
    };

    lettore.readAsText(file);
}
</script>
